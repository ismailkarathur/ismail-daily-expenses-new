async function saveTransaction(e) { e.preventDefault(); const saveBtn = document.getElementById('saveEntryBtn'); // Prevent double click / duplicate entries if (saveBtn.disabled) return; saveBtn.disabled = true; saveBtn.innerText = 'Saving...'; try { const id = document.getElementById('editDocId').value; const title = document.getElementById('txTitle').value.trim(); const amount = parseFloat(document.getElementById('txAmount').value); const category = document.getElementById('txCategory').value; const date = document.getElementById('txDate').value; const type = document.getElementById('txType').value; const site = type === 'Expense' ? document.getElementById('txSite').value.trim() : ''; const fileInput = document.getElementById('txFile'); const file = fileInput.files[0]; if (!title || !amount || amount <= 0 || !category || !date) { alert('Please fill all required fields.'); return; } const data = { title, amount, category, date, site, type, uid: currentUser.uid, updatedAt: new Date() }; // -------------------------------------------------- // EDIT EXISTING ENTRY // -------------------------------------------------- if (id) { // Close immediately so UI doesn't feel stuck closeTransactionModal(); // Update UI immediately const index = allTx.findIndex(x => x.id === id); if (index !== -1) { allTx[index] = { ...allTx[index], ...data }; updateSummary(); renderTable(); renderChart(); } // Save to Firebase in background (async () => { try { if (file) { const storageRef = storage.ref( `receipts/${currentUser.uid}/${Date.now()}_${file.name}` ); const snapshot = await storageRef.put(file); data.fileUrl = await snapshot.ref.getDownloadURL(); } await db.collection('expenses').doc(id).update(data); } catch (err) { console.error('Background update error:', err); alert('Entry updated on screen, but Firebase sync failed: ' + err.message); // Refresh from Firebase listenToData(); } })(); return; } // -------------------------------------------------- // NEW ENTRY // -------------------------------------------------- // Generate document ID first const newDocRef = db.collection('expenses').doc(); const newId = newDocRef.id; const optimisticEntry = { id: newId, ...data }; // Add immediately to local UI allTx.unshift(optimisticEntry); updateSummary(); renderTable(); renderChart(); // Close immediately closeTransactionModal(); // Save in background (async () => { try { // Upload attachment only if selected if (file) { const storageRef = storage.ref( `receipts/${currentUser.uid}/${Date.now()}_${file.name}` ); const snapshot = await storageRef.put(file); const fileUrl = await snapshot.ref.getDownloadURL(); data.fileUrl = fileUrl; // Update local UI with file link const localIndex = allTx.findIndex(x => x.id === newId); if (localIndex !== -1) { allTx[localIndex].fileUrl = fileUrl; renderTable(); } } // Firebase save await newDocRef.set(data); console.log('Transaction saved successfully:', newId); } catch (err) { console.error('Background save error:', err); // Remove optimistic entry if Firebase save failed allTx = allTx.filter(x => x.id !== newId); updateSummary(); renderTable(); renderChart(); alert( 'Transaction could not be saved to Firebase.\n\n' + err.message ); } })(); } catch (err) { console.error('Save error:', err); alert('Error: ' + err.message); } finally { saveBtn.disabled = false; saveBtn.innerText = 'Save Entry'; } }
